cmake_minimum_required(VERSION 3.16)
project(concurrent_astar CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)
set(LOCK_INC "${CMAKE_SOURCE_DIR}/include" CACHE PATH "directory containing the Assignment-1 lock.h")

add_executable(bench src/bench_main.cpp)
add_executable(bench_pareto src/bench_pareto.cpp)
add_executable(test_astar tests/test_astar.cpp)
add_executable(test_pareto tests/test_pareto.cpp)
foreach(t bench bench_pareto test_astar test_pareto)
  target_include_directories(${t} PRIVATE include ${LOCK_INC})
  target_link_libraries(${t} PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W3 /O2 /permissive-)
    target_compile_definitions(${t} PRIVATE _USE_MATH_DEFINES NOMINMAX)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -O2 -g)
  endif()
endforeach()
if(WIN32)
  target_link_libraries(bench PRIVATE psapi)
  target_link_libraries(bench_pareto PRIVATE psapi)
endif()
